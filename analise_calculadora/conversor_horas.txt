conversor horas

import { useState } from "react";

export default function TimeConverter() {
  const [decimalTime, setDecimalTime] = useState("");
  const [convertedTime, setConvertedTime] = useState("");

  const convertToHoursAndMinutes = (decimal) => {
    if (isNaN(decimal) || decimal === "") return "";
    
    const totalMinutes = Math.round(decimal * 60);
    const hours = Math.floor(totalMinutes / 60);
    const minutes = totalMinutes % 60;

    return `${hours}h ${minutes}min`;
  };

  const handleChange = (e) => {
    const value = e.target.value;
    setDecimalTime(value);
    setConvertedTime(convertToHoursAndMinutes(parseFloat(value)));
  };

  return (
    <div className="flex flex-col items-center justify-center h-screen bg-gray-100 p-4">
      <h1 className="text-2xl font-bold mb-4">Conversor de Horas Decimais</h1>
      <input
        type="number"
        step="0.01"
        placeholder="Digite um número decimal"
        value={decimalTime}
        onChange={handleChange}
        className="p-2 border border-gray-300 rounded-lg mb-4 text-center"
      />
      <div className="text-xl font-semibold text-blue-600">{convertedTime}</div>
    </div>
  );
}
